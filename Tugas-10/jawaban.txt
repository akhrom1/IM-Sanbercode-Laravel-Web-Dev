2. Tabel create User | categories | Items

create table users (
  id INT PRIMARY KEY AUTO_INCREMENT,
  name varchar(225),
  email varchar(225),
  password varchar(225)
)

create table categories (
  id INT PRIMARY KEY AUTO_INCREMENT,
  name varchar(225)
)

create table items (
  id INT PRIMARY KEY AUTO_INCREMENT,
  name varchar(225),
  description varchar(255),
  price int,
  stock int,
  category_id int,
  foreign key (category_id) references categories(id)
)
